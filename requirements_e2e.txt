# E2E test dependencies — separate from tests/requirements.txt (fast, HA-free
# unit tests) so the unit-test setup never needs a browser runtime.
#
# Used by Dockerfile.e2e (Playwright runner image) and by local runs:
#   pip install -r requirements_e2e.txt && playwright install chromium

# Browser driver + pytest integration (fixtures `page`, `context`, ...).
# Pinned EXACTLY to the Playwright version baked into Dockerfile.e2e's base
# image (mcr.microsoft.com/playwright/python:v1.47.0-jammy). A range here lets
# pip resolve a newer driver than the image's browser binaries, which makes the
# run fail with a headless-shell mismatch that looks like an app regression.
# Bump together with Dockerfile.e2e, never alone.
playwright==1.47.0
pytest-playwright>=0.5

# scripts/e2e_bootstrap.py: HA onboarding / config-entry / options-flow REST
# calls.
requests>=2.32

# scripts/e2e_bootstrap.py: reading the person list (stable slug) and clearing
# the per-user UI selection need HA's WebSocket API — `health_o_mat/list_persons`
# and `health_o_mat/set_ui_settings` are websocket-only commands
# (custom_components/health_o_mat/ws_api.py), there is no REST equivalent.
websockets>=13.0

# tests_e2e/test_lab_values_e2e.py parses examples/dashboard-labs-card.yaml
# (the Lovelace dashboard artifact under test, REQ-HOM-126) via
# `yaml.safe_load`. Neither the Playwright base image nor the unit-test
# requirements ship PyYAML, so without this pin the import fails at collection.
pyyaml>=6
